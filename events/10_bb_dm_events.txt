namespace = bb_gm_events

bb_gm_events.1 = {
	type = character_event
	title = bbgm_make_custom_announcement
	desc = bbgm_make_custom_announcement_desc
	theme = diplomacy
	override_background = { reference = physicians_study }
	left_portrait = {
		character = root
		animation = chancellor
	}

	widget = {
		gui = "mod_event_window_widget_custom_announcement"
		container = "custom_widgets_container"
		controller = {
			type = text
			data = {
				key = customannouncement_text
			}
		}
		setup_scope = {
			root = { save_scope_as = text_target }
		}
	}
	option = { 
		name = bb_gm_custom_announcement.1
	}

	after = {
		scope:actor = {
			store_localized_text_in_death = customannouncement_text
		}
		every_player = {
			send_interface_message = {
				type = bbgm_custom_announcement_alert
				left_icon = scope:actor
			}
		}
		remove_localized_text = customannouncement_text
	}
}

baseborn_create_title.1002 = {
	hidden = yes
		
	immediate = {
		create_dynamic_title = {
			tier = duchy
			name = GET_TARGET_TITLE_NAME
		}

		create_title_and_vassal_change = {
			type = created
			save_scope_as = change
			add_claim_on_loss = no
		}
		scope:new_title = {
			set_color_from_title = scope:target
			set_coa = scope:target
			change_title_holder = {
				holder = root
				change = scope:change
			}
		}

		resolve_title_and_vassal_change = scope:change

		scope:new_title = {
			set_global_variable = {
				name = dm_tool_created_title
				value = scope:new_title
			}
		}
		every_player = {
			send_interface_message = {
				type = dm_tool_alert_created_title
				left_icon = root
				seconds = 10
			}
		}
	}
}
