bbgm_become_game_master = {
    category = interaction_category_bb_game_master
    interface_priority = 0
    icon = revoke_governorship
    desc = bbgm_become_game_master_desc
    auto_accept = yes
    use_diplomatic_range = no
    ignores_pending_interaction_block = yes

    is_shown = {
        OR = {
            scope:actor = { has_trait = game_master }
            debug_only = yes
        }
        scope:recipient = {
            NOT = { has_trait = game_master }
        }
    }

    on_send = {
        scope:actor = {
            remove_trait = game_master
            set_player_character = scope:recipient
            house = { 
                reset_coa = yes
                set_house_name_from_house = root.house.dynasty
            }
            death = { death_reason = death_vanished }
        }
        scope:recipient = {
            clear_traits = yes
            add_trait = game_master
            set_character_faith = faith:fg_pan_bronze
            change_first_name = The_Game_Master
            remove_nickname = yes
            divorce = primary_spouse
            primary_title = { set_coa = bb_dm_coa }
            house = {
                set_house_name = dynn_GM
                set_coa = bb_dm_coa
            }
        }
    }
}

bbgm_independence = {
    category = interaction_category_bb_game_master
    interface_priority = 96
    icon = icon_found_despotate
    desc = bbgm_independence_desc
    auto_accept = yes
    use_diplomatic_range = no
    ignores_pending_interaction_block = yes

    is_shown = {
		NOT = { scope:recipient = scope:actor }
        scope:actor = {
            is_ai = no
            has_trait = game_master
        }
        scope:recipient = { is_independent_ruler = no }
    }

    on_send = {
		create_title_and_vassal_change = {
			type = independency
			save_scope_as = change
		}
        scope:recipient = {
            change_liege_or_become_independent = {
                CHANGE = scope:change
                VASSAL = this
            }
        }
		resolve_title_and_vassal_change = scope:change
    }
}
