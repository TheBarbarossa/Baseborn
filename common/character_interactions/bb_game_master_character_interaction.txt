bbgm_become_game_master = {
    category = interaction_category_bb_game_master
    interface_priority = 0
    icon = revoke_governorship
    desc = bbgm_become_game_master_desc
    auto_accept = yes
    use_diplomatic_range = no
    ignores_pending_interaction_block = yes
	common_interaction = yes

    is_shown = {
        OR = {
            scope:actor = { has_trait = game_master }
            debug_only = yes
        }
        scope:recipient = {
            NOT = { has_trait = game_master }
        }
    }

    on_send = {
        scope:actor = {
            remove_trait = game_master
            set_player_character = scope:recipient
            house = { 
                reset_coa = yes
                set_house_name_from_house = root.house
            }
            death = { death_reason = death_vanished }
        }
        scope:recipient = {
            clear_traits = yes
            add_trait = game_master
            set_character_faith = faith:fg_pan_bronze
            change_first_name = The_Game_Master
            remove_nickname = yes
            every_spouse = { divorce = prev }
            primary_title = { set_coa = bb_dm_coa }
            house = {
                set_house_name = dynn_GM
                set_coa = bb_dm_coa
            }
        }
    }
}

bbgm_start_pregnancy = {
    category = interaction_category_bb_game_master
    interface_priority = 97
    icon = child
    desc = bbgm_start_pregnancy_desc
    auto_accept = yes
    use_diplomatic_range = no
    ignores_pending_interaction_block = yes
	common_interaction = yes

    is_shown = {
		NOT = { scope:recipient = scope:actor }
        scope:recipient = {
            is_female = yes
            is_pregnant = no
        }
        scope:actor = {
            is_ai = no
            has_trait = game_master
        }
    }

    populate_actor_list = {
        every_living_character = {
            limit = { has_relation_lover = scope:recipient }
            add_to_list = characters
        }
        scope:recipient = {
            every_spouse = { add_to_list = characters }
        }
    }

	can_be_picked = {
        always = yes
	}

    on_send = {
        scope:recipient = {
            make_pregnant = { father = scope:secondary_actor }
            #AGOT Added; LMF
            event_pregnancy_effect = yes
        }
    }
}
