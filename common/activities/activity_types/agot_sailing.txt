activity_agot_sailing = {
	is_shown = {
		OR = {
			AND = {
				is_landed = yes
				any_held_county = {
					any_county_province = {
						OR = {
							has_building_or_higher = common_tradeport_01
							has_sailing_special_building = yes
							has_building_or_higher = agot_great_fleet_01
						}
					}
				}
			}
			is_landless_adventurer = yes
			highest_held_title_tier >= tier_kingdom
			government_has_flag = government_is_pirate_trigger_check
		}
		NOT = { government_has_flag = government_is_nw }
		NOT = { government_has_flag = government_is_uninteractable }
	}

	can_start_showing_failures_only = {
		is_available_healthy_adult = yes
		is_at_war = no
		trigger_if = {
			limit = { is_landed = yes }
			custom_tooltip = {
				text = realm_is_isolated
				NOR = {
					any_liege_or_above = {
						capital_county = { has_county_modifier = agot_containing_the_spread }
					}
					capital_county = { has_county_modifier = agot_containing_the_spread }
				}
			}
		}
		trigger_else = {
			domicile.domicile_location.county.holder = {
				custom_tooltip = {
					text = realm_is_isolated
					NOR = {
						any_liege_or_above = {
							capital_county = { has_county_modifier = agot_containing_the_spread }
						}
						capital_county = { has_county_modifier = agot_containing_the_spread }
					}
				}
			}
		}
	}

	is_valid = {
		scope:host = {
			is_alive = yes
			is_imprisoned = no
			NOT = { is_incapable = yes }
		}
		scope:host = {
			OR = {
				agot_sailing_type_hunt_requirements = yes
				agot_sailing_type_raid_requirements = yes
				agot_sailing_type_treasure_requirements = yes
				agot_sailing_type_valyria_requirements = yes
			}
		}
	}

	on_invalidated = { #
		# Save scopes for flavor events
		scope:activity = {
			activity_location = { save_scope_as = location }
		}
		if = {
			limit = {
				scope:host = { is_incapable = yes }
			}
			scope:activity = {
				activity_type = { save_scope_as = activity_type }
			}
			scope:host = { trigger_event = activity_system.0330 }
			every_attending_character = {
				limit = {
					NOT = { this = scope:host }
				}
				trigger_event = activity_system.0331
			}
		}
		scope:host = {
			if = {
				limit = { has_character_flag = treasure_hunt }
				remove_character_flag = treasure_hunt
			}
			if = {
				limit = { has_character_flag = traveling_to_valyria }
				remove_character_flag = traveling_to_valyria
			}
		}
	}

	cost = {
		gold = {
			add = {
				add = {
					value = sailing_base_cost #Base level
					desc = hunt_base_cost
				}
				add = {
					value = sailing_base_cost
					multiply = activity_cost_scale_by_era
					subtract = sailing_base_cost
					desc = activity_cost_scale_by_era_desc
				}
				if = {
					limit = {
						dynasty ?= { has_dynasty_perk = law_legacy_1 }
					}
					subtract = {
						value = 0
						add = {
							value = sailing_base_cost
							add = {
								value = sailing_base_cost
								multiply = activity_cost_scale_by_era
								subtract = sailing_base_cost
							}
						}
						multiply = law_legacy_cost_reduction_mult
						desc = law_legacy_1_name
					}
				}
				if = {
					limit = {
						scope:province ?= { has_building_or_higher = agot_great_fleet_01 }
					}
					multiply = {
						value = 0.6
						desc = activity_agot_sailing_province_desc_great_fleet
					}
				}
				if = {
					limit = {
						scope:province ?= { has_building_or_higher = common_tradeport_07 }
					}
					multiply = {
						value = 0.6
						desc = activity_agot_sailing_province_desc_common_tradeport_max
					}
				}
				else_if = {
					limit = {
						scope:province ?= { has_building = common_tradeport_06 }
					}
					multiply = {
						value = 0.7
						desc = activity_agot_sailing_province_desc_common_tradeport_major
					}
				}
				else_if = {
					limit = {
						scope:province ?= { has_building_or_higher = common_tradeport_03 }
					}
					multiply = {
						value = 0.8
						desc = activity_agot_sailing_province_desc_common_tradeport_medium
					}
				}
				else_if = {
					limit = {
						scope:province ?= { has_building_or_higher = common_tradeport_01 }
					}
					multiply = {
						value = 0.9
						desc = activity_agot_sailing_province_desc_common_tradeport_minor
					}
				}
				if = {
					limit = {
						scope:province.county.holder ?= {
							employs_court_position = harbormaster_court_position
						}
					}
					if = {
						limit = {
							scope:province.county.holder ?= {
								harbormaster_court_position_apptitude_check = { VALUE = 3 }
							}
						}
						multiply = {
							value = 0.8
							desc = activity_agot_sailing_harbormaster_court_position_apptitude_check_3
						}
					}
					else_if = {
						limit = {
							scope:province.county.holder ?= {
								harbormaster_court_position_apptitude_check = { VALUE = 4 }
							}
						}
						multiply = {
							value = 0.7
							desc = activity_agot_sailing_harbormaster_court_position_apptitude_check_4
						}
					}
					else_if = {
						limit = {
							scope:province.county.holder ?= {
								harbormaster_court_position_apptitude_check = { VALUE = 5 }
							}
						}
						multiply = {
							value = 0.6
							desc = activity_agot_sailing_harbormaster_court_position_apptitude_check_5
						}
					}
				}
				if = {
					limit = {
						scope:province ?= { has_sailing_special_building = yes }
					}
					multiply = {
						value = 0.6
						desc = activity_agot_sailing_province_desc_special_building
					}
				}
			}
		}
	}

	ui_predicted_cost = {
		# All costs are balanced on County/Early Era starting point
		gold = {
			add = { # Base cost, scaled by era
				value = sailing_base_cost
				multiply = activity_cost_scale_by_era
			}
			if = {
				limit = {
					dynasty ?= { has_dynasty_perk = law_legacy_1 }
				}
				subtract = {
					value = sailing_base_cost
					multiply = law_legacy_cost_reduction_mult
				}
			}
			# Make it a multiple of 5 (rounded up)
			divide = 5
			ceiling = yes
			multiply = 5
		}
	}

	on_host_death = { invalidate_activity = scope:activity }

	is_location_valid = {
		save_temporary_scope_as = location_in_question
		trigger_if = {
			limit = {
				NOT = { scope:special_option ?= flag:agot_sailing_type_treasure }
			}
			trigger_if = {
				limit = {
					scope:host = { is_landed = yes }
				}
				# Only display plague if the location is otherwise valid
				trigger_if = {
					limit = {
						agot_sailing_location_is_valid = yes
					}
					custom_tooltip = {
						text = realm_is_isolated
						NOT = {
							duchy ?= {
								any_de_jure_county = { has_county_modifier = agot_containing_the_spread }
							}
						}
					}
				}
				trigger_else = {
					custom_tooltip = {
						text = agot_sailing_province_requirement_landed
						always = no
					}
				}
			}
			trigger_else = { # Landless characters
				# Only display plague if the location is otherwise valid
				trigger_if = {
					limit = {
						agot_sailing_location_is_valid_landless = yes
					}
					custom_tooltip = {
						text = realm_is_isolated
						NOT = {
							duchy ?= {
								any_de_jure_county = { has_county_modifier = agot_containing_the_spread }
							}
						}
					}
				}
				trigger_else = {
					custom_tooltip = {
						text = agot_sailing_province_requirement_landless
						always = no
					}
				}
			}
		}
		trigger_else = {
			trigger_if = {
				limit = {
					OR = {
						scope:host = {
							has_map_here = { LOCATION = scope:location_in_question }
						}
						has_province_modifier = sunken_ship_sighted
					}
				}
				always = yes
			}
			trigger_else = { always = no }
		}
	}

	ai_will_do = {
		value = 50

		if = {
			limit = { agot_prefers_sailing = yes }
			add = 40
		}
		if = {
			limit = { highest_held_title_tier = tier_kingdom }
			add = -30
		}
		else_if = {
			limit = { highest_held_title_tier = tier_empire }
			add = -40
		}
		add = {
			value = ai_sociability
			multiply = 0.5
			desc = "ai_sociability"
		}
		if = {
			limit = { ai_energy < 0 }
			add = {
				add = ai_energy
				desc = "ai_energy"
			}
		}
		if = {
			limit = {
				any_held_county = {
					any_county_province = { is_coastal = no }
				}
			}
			add = -100
		}
		if = {
			limit = {
				any_held_county = {
					any_county_province = { has_building_or_higher = agot_great_fleet_01 }
				}
			}
			add = 35
		}
		# Personality
		if = { # Avoids responsibilities
			limit = { has_trait = lazy }
			add = {
				value = 20
				desc = lazy
			}
		}
		else_if = { # Better things to do
			limit = { has_trait = diligent }
			add = {
				value = -10
				desc = diligent
			}
		}
		if = { # Likes hurting things
			limit = { has_trait = sadistic }
			add = {
				value = 20
				desc = sadistic
			}
		}
		else_if = { # Empathetic
			limit = { has_trait = compassionate }
			add = {
				value = -10
				desc = compassionate
			}
		}
		if = { # Likes to socialize
			limit = { has_trait = gregarious }
			add = {
				value = 20
				desc = gregarious
			}
		}
		else_if = {
			limit = { has_trait = shy }
			add = {
				value = -10
				desc = shy
			}
		}
		if = {
			limit = { has_trait = reclusive }
			add = {
				value = -25
				desc = reclusive
			}
		}
		# Culture
		if = {
			limit = {
				culture = { has_cultural_tradition = tradition_seafaring }
			}
			add = 20
		}
		# Court
		if = {
			limit = {
				has_royal_court = yes
				has_court_type = court_warlike
			}
			add = {
				value = 15
				desc = court_warlike
			}
		}
		# Education
		if = {
			limit = { has_trait = education_martial }
			add = {
				value = 20
				desc = education_martial
			}
		}
		# Prestige
		if = {
			limit = { prestige < 0 }
			add = {
				value = 30
				desc = "Very low prestige"
			}
		}
		if = {
			limit = { prestige_level <= 1 }
			add = {
				value = 5
				desc = "Low prestige level"
			}
		}
		# Lifestyle
		if = {
			limit = { has_trait = lifestyle_mariner }
			add = {
				value = 25
				desc = lifestyle_mariner
			}
		}
		if = {
			limit = {
				has_trait = lifestyle_mariner
				has_trait_xp = {
					trait = lifestyle_mariner
					value >= trait_second_level
				}
			}
			add = {
				value = 25
				desc = "experienced hunter"
			}
		}
		# Stress
		if = { # Low Stress
			limit = { stress >= lower_than_baseline_stress }
			add = {
				value = 20
				desc = "lower_than_baseline_stress"
			}
		}
		if = { # High Stress
			limit = { stress >= higher_than_baseline_stress }
			add = {
				value = 20
				desc = "higher_than_baseline_stress"
			}
		}
		if = { # Very High Stress
			limit = { stress >= high_stress }
			add = {
				value = 20
				desc = "high_stress"
			}
		}
		if = {
			limit = {
				short_term_gold >= hunt_activity_base_cost
				short_term_gold >= major_gold_value
			}
			add = {
				value = 30
				desc = "Got gold to burn"
			}
		}
		if = {
			limit = { has_character_flag = hunts_a_lot }
			add = {
				value = 500
				desc = "hunts a lot"
			}
		}
		if = {
			limit = {
				ai_has_conqueror_personality = yes
				OR = {
					AND = {
						prestige_level < 4
						can_use_conquest_cbs_trigger = yes
					}
					prestige < 1000
				}
			}
			add = {
				value = 1000
				desc = "conqueror looking for prestige"
			}
		}
		if = {
			limit = {
				ai_has_conqueror_personality = yes
				trigger_if = {
					limit = {
						prestige_level >= 4
						can_use_conquest_cbs_trigger = yes
					}
					prestige_level >= 4
				}
				prestige >= 1000
			}
			add = {
				value = -5000
				desc = "conqueror sated for prestige"
			}
		}
		# Legitimacy
		if = {
			limit = {
				is_valid_for_legitimacy_change = yes
			}
			if = { # Low Legitimacy
				limit = {
					OR = {
						legitimacy_level = 1
						legitimacy_level = 2
					}
				}
				add = {
					value = 10
					desc = "low_legitimacy"
				}
			}
			if = { # Very Low Legitimacy
				limit = {
					legitimacy_level = 0
				}
				add = {
					value = 20
					desc = "low_legitimacy"
				}
			}
		}
		# Plague
		if = {
			limit = {
				any_held_county = {
					any_county_province = {
						any_province_epidemic = {
							count >= 1
							outbreak_intensity < apocalyptic
						}
					}
				}
				would_follow_social_distancing_value >= -1
			}
			add = {
				value = -60
				desc = "Plague in Domain"
			}
		}
		else_if = {
			limit = {
			any_held_county = {
				any_county_province = {
						any_province_epidemic = {
							count >= 1
							outbreak_intensity < apocalyptic
						}
					}
				}
				would_follow_social_distancing_value >= -1
			}
			add = {
				value = -20
				desc = "Plague in Top Realm"
			}
		}
		if = {
			limit = {
				any_held_county = {
					any_county_province = {
						any_province_epidemic = {
							count >= 1
							outbreak_intensity = apocalyptic
						}
					}
				}
			}
			add = {
				value = -200
				desc = "Apocalyptic Plague in Domain"
			}
		}
		else_if = {
			limit = {
				any_held_county = {
					any_county_province = {
						any_province_epidemic = {
							count >= 1
							outbreak_intensity = apocalyptic
						}
					}
				}
			}
			add = {
				value = -40
				desc = "Apocalyptic Plague in Top Realm"
			}
		}
		if = { # Has a ship
			limit = {
				has_access_to_a_ship = yes
			}
			add = {
				value = 50
			}
		}
	}
